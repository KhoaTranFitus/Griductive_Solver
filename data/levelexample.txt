 class ClueType(str, Enum):
    NONE = "NONE"

    FACT = "FACT"

    SAME = "SAME"
    DIFFERENT = "DIFFERENT"

    EXACTLY = "EXACTLY"
    AT_LEAST = "AT_LEAST"
    AT_MOST = "AT_MOST"

    EQUAL_COUNT = "EQUAL_COUNT"
    COMPARE_COUNT = "COMPARE_COUNT"

    PARITY = "PARITY"
    CONNECTED = "CONNECTED"

    COUNT_PROPERTY = "COUNT_PROPERTY"



class RegionType(str, Enum):
    ROW = "ROW"
    COLUMN = "COLUMN"

    NEIGHBORS = "NEIGHBORS"

    EDGES = "EDGES"

    LEFT_OF = "LEFT_OF"
    RIGHT_OF = "RIGHT_OF"
    ABOVE = "ABOVE"
    BELOW = "BELOW"

    EXPLICIT = "EXPLICIT"

    INTERSECTION = "INTERSECTION"
    UNION = "UNION"
